## Go 1.21 (August 2023)

### Built-in Functions: min, max, clear

```go
// min and max work with any ordered type
x := min(1, 2, 3)                    // 1
y := max(1.5, 2.7, 1.1)              // 2.7
s := min("apple", "banana", "cherry") // "apple"

// Multiple arguments
func clamp[T cmp.Ordered](v, lo, hi T) T {
    return max(lo, min(v, hi))
}

// clear for maps and slices
m := map[string]int{"a": 1, "b": 2}
clear(m) // m is now empty but allocated

s := []int{1, 2, 3, 4, 5}
clear(s) // s is now []int{0, 0, 0, 0, 0}
```

### log/slog Package

Structured logging standard library.

```go
import "log/slog"

func main() {
    // Default text handler
    slog.Info("user login",
        "user_id", 12345,
        "ip", "192.168.1.1",
        "duration_ms", 234,
    )
    // Output: 2023/08/01 12:34:56 INFO user login user_id=12345 ip=192.168.1.1 duration_ms=234

    // JSON handler for production
    logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
    logger.Info("request processed",
        slog.String("method", "GET"),
        slog.Int("status", 200),
        slog.Duration("latency", 45*time.Millisecond),
    )
    // Output: {"time":"2023-08-01T12:34:56.789","level":"INFO","msg":"request processed","method":"GET","status":200,"latency":45000000}

    // Grouped attributes
    logger.Info("database query",
        slog.Group("db",
            slog.String("host", "localhost"),
            slog.Int("port", 5432),
        ),
        slog.Duration("query_time", 12*time.Millisecond),
    )

    // With context
    logger = logger.With(
        slog.String("service", "api"),
        slog.String("version", "1.2.3"),
    )
    logger.Info("starting") // Includes service and version in all logs
}
```

### slices Package

Generic slice operations.

```go
import "slices"

func main() {
    s := []int{3, 1, 4, 1, 5, 9}

    // Sorting
    slices.Sort(s)
    fmt.Println(s) // [1 1 3 4 5 9]

    // Binary search (requires sorted slice)
    idx, found := slices.BinarySearch(s, 4)
    fmt.Println(idx, found) // 3 true

    // Contains
    if slices.Contains(s, 5) {
        fmt.Println("Found 5")
    }

    // Compare
    s2 := []int{1, 1, 3, 4, 5, 9}
    fmt.Println(slices.Equal(s, s2)) // true

    // Compact (remove consecutive duplicates)
    s3 := []int{1, 1, 2, 3, 3, 3, 4}
    s3 = slices.Compact(s3)
    fmt.Println(s3) // [1 2 3 4]

    // Clone
    s4 := slices.Clone(s)

    // Delete
    s5 := []int{0, 1, 2, 3, 4}
    s5 = slices.Delete(s5, 1, 3) // Remove indices [1:3)
    fmt.Println(s5) // [0 3 4]

    // Insert
    s6 := []int{1, 2, 5}
    s6 = slices.Insert(s6, 2, 3, 4)
    fmt.Println(s6) // [1 2 3 4 5]
}
```

### maps Package

Generic map operations.

```go
import "maps"

func main() {
    m1 := map[string]int{"a": 1, "b": 2}

    // Clone
    m2 := maps.Clone(m1)

    // Copy
    m3 := map[string]int{"c": 3}
    maps.Copy(m3, m1) // m3 now has a, b, c

    // Delete func
    m4 := map[string]int{"a": 1, "b": 2, "c": 3, "d": 4}
    maps.DeleteFunc(m4, func(k string, v int) bool {
        return v%2 == 0 // Delete even values
    })
    fmt.Println(m4) // map[a:1 c:3]

    // Equal
    m5 := map[string]int{"a": 1, "b": 2}
    fmt.Println(maps.Equal(m1, m5)) // true

    // EqualFunc (custom comparison)
    m6 := map[string]float64{"a": 1.0, "b": 2.0}
    m7 := map[string]float64{"a": 1.1, "b": 2.1}
    equal := maps.EqualFunc(m6, m7, func(v1, v2 float64) bool {
        return math.Abs(v1-v2) < 0.2
    })
    fmt.Println(equal) // true
}
```

### cmp Package

Generic comparison utilities.

```go
import "cmp"

// Ordered constraint
func Max[T cmp.Ordered](a, b T) T {
    if a > b {
        return a
    }
    return b
}

// Compare function
func CompareInts(a, b int) int {
    return cmp.Compare(a, b) // Returns -1, 0, or 1
}

// Or function (return first non-zero value)
func FirstNonEmpty(values ...string) string {
    for _, v := range values {
        if v := cmp.Or(v, ""); v != "" {
            return v
        }
    }
    return ""
}

// Usage
fmt.Println(cmp.Compare(5, 10))    // -1
fmt.Println(cmp.Compare(10, 5))    // 1
fmt.Println(cmp.Compare(5, 5))     // 0
fmt.Println(cmp.Or("", "", "hi"))  // "hi"
```
